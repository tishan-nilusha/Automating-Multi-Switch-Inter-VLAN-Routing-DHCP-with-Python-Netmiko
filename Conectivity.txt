Network Connectivity Build Configurations

1. Ubuntu PC Route Configuration
--------------------------------
# Add static route to reach the internal network segments via Router 1
ip route add 10.10.0.0/16 via 192.168.42.2

2. Core Router Configuration
--------------------------------
Router 1 (R1)
conf t
interface ethernet 0/3
 ip address 192.168.42.2 255.255.255.0
 no shutdown
exit
interface ethernet 0/0
 ip address 10.10.12.1 255.255.255.0
 no shutdown
exit
interface ethernet 0/1
 ip address 10.10.13.1 255.255.255.0
 no shutdown
exit
interface ethernet 0/2
 ip address 10.10.14.1 255.255.255.0
 no shutdown
exit
interface loopback 0
 ip address 1.1.1.1 255.255.255.255
exit
ip domain-name cisco.local
crypto key generate rsa modulus 2048
username admin privilege 15 secret cisco
line vty 0 4
 transport input ssh
 login local
exit
router ospf 1
 network 1.1.1.1 0.0.0.0 area 0
 network 192.168.42.0 0.0.0.255 area 0
 network 10.10.12.0 0.0.0.255 area 0
 network 10.10.13.0 0.0.0.255 area 0
 network 10.10.14.0 0.0.0.255 area 0
end
